# Paste this into .github/workflows/test-app-android.yml (kept here as a reference).
name: App Automate - Android

on:
  workflow_dispatch:
    inputs:
      username:
        description: BrowserStack username
        required: true
      accessKey:
        description: BrowserStack access key
        required: true
      localTesting:
        description: Use BrowserStack Local
        required: false
        default: "false"
      features:
        description: Comma-separated features; empty runs the full flow
        required: false
        default: ""

jobs:
  android:
    runs-on: ubuntu-latest
    env:
      BROWSERSTACK_USERNAME: ${{ inputs.username }}
      BROWSERSTACK_ACCESS_KEY: ${{ inputs.accessKey }}
      BROWSERSTACK_LOCAL: ${{ inputs.localTesting }}
      FEATURES: ${{ inputs.features }}
      # Test data for the app (set as repository secrets/variables)
      REGISTER_NAME: ${{ secrets.APP_REGISTER_NAME }}
      REGISTER_EMAIL: ${{ secrets.APP_REGISTER_EMAIL }}
      REGISTER_PASSWORD: ${{ secrets.APP_REGISTER_PASSWORD }}
      PHONE_NUMBER: ${{ secrets.APP_PHONE_NUMBER }}
      OTP_VALUE: ${{ secrets.APP_OTP_VALUE }}
    steps:
      - uses: actions/checkout@v4

      - name: Mask credentials
        shell: bash
        run: |
          echo "::add-mask::${BROWSERSTACK_USERNAME}"
          echo "::add-mask::${BROWSERSTACK_ACCESS_KEY}"

      # =========================================================
      # Verify APK
      # =========================================================
      - name: Verify APK
        shell: bash
        run: |
          set +x
          APK_PATH="${{ github.workspace }}/docs/assets/NexusDemo-v2.9.10.apk"
          if [ ! -f "$APK_PATH" ]; then
            echo "APK not found:"
            echo "$APK_PATH"
            exit 1
          fi
          echo "APK verified successfully."

      # =========================================================
      # Upload APK to BrowserStack
      # =========================================================
      - name: Upload APK to BrowserStack
        id: upload
        shell: bash
        run: |
          set +x
          APK_PATH="${{ github.workspace }}/docs/assets/NexusDemo-v2.9.10.apk"
          RESPONSE=$(
            curl -sS \
              --user "${BROWSERSTACK_USERNAME}:${BROWSERSTACK_ACCESS_KEY}" \
              --request POST \
              "https://api-cloud.browserstack.com/app-automate/upload" \
              --form "file=@${APK_PATH}" \
              --form "custom_id=SampleApp"
          )
          APP_URL=$(printf '%s' "$RESPONSE" | jq -r '.app_url // empty')
          if [ -z "$APP_URL" ]; then
            echo "BrowserStack APK upload failed."
            ERROR_MSG=$(printf '%s' "$RESPONSE" | jq -r '.error // empty')
            if [ -n "$ERROR_MSG" ]; then
              echo "BrowserStack returned an upload error."
            else
              echo "BrowserStack returned an unexpected response."
            fi
            exit 1
          fi
          echo "::add-mask::$APP_URL"
          echo "APP_ID=$APP_URL" >> "$GITHUB_OUTPUT"
          echo "APK uploaded successfully."

      - uses: actions/setup-node@v4
        with:
          node-version: 20

      - name: Install dependencies
        working-directory: app-tests
        run: npm install

      # FEATURES empty => specs/full-flow.spec.js; otherwise one spec per selected feature
      - name: Run App Automate tests
        working-directory: app-tests
        env:
          BROWSERSTACK_APP_URL: ${{ steps.upload.outputs.APP_ID }}
        run: npm test
